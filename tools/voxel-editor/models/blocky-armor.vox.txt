# The blocky character in armor: overlays are pixel layers on a part, painted one face per pixel.
#   tools/voxel-editor/vox run models/blocky-armor.j3voxel --script models/blocky-armor.vox.txt --render armor.png --view all
# `grow 1` puts each layer one voxel out from its part, so the grid gets a one voxel margin first.
open models/blocky
size 18 34 10
shift 1 1 1
color 9 #aeb6c2 steel
color 10 #e0b43c gold
color 11 #565d68 darksteel
color 12 #b3262e crimson

# Helmet with a visor slit and a gold crest. Faces are px nx py ny pz nz: the +x side, -x side, and so on.
overlay new helmet Head grow 1
overlay box helmet 0 0 0 17 33 9 all steel
overlay box helmet 8 33 0 9 33 9 py gold
overlay erase helmet 4 28 9 13 29 9 pz

# Chestplate with a gold belt line and a crimson emblem on the front.
overlay new chest Torso grow 1
overlay box chest 0 0 0 17 33 9 all steel
overlay box chest 4 14 0 13 14 9 all gold
overlay box chest 8 18 0 9 21 9 pz crimson

# Pauldrons on both upper arms.
overlay new pauldronL UpperArm.L grow 1
overlay box pauldronL 0 0 0 17 33 9 all steel
overlay box pauldronL 0 25 0 17 25 9 py gold
overlay new pauldronR UpperArm.R grow 1
overlay box pauldronR 0 0 0 17 33 9 all steel
overlay box pauldronR 0 25 0 17 25 9 py gold

# Boots: the shell covers the whole lower leg, so erase everything above the ankle.
overlay new bootL LowerLeg.L grow 1
overlay box bootL 0 0 0 17 33 9 all darksteel
overlay erase bootL 0 5 0 17 33 9
overlay new bootR LowerLeg.R grow 1
overlay box bootR 0 0 0 17 33 9 all darksteel
overlay erase bootR 0 5 0 17 33 9
rig check
overlay list
pose load wave_a
