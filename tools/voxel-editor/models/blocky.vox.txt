# A Minecraft-style character, rigged. Rebuild and render:
#   tools/voxel-editor/vox run models/blocky.j3voxel --script models/blocky.vox.txt --render blocky.png --view all
# 1 voxel = 1 "pixel": head 8x8x8, body 8x12x4, arms and legs 4x12x4. Faces +z, left (.L) is +x.
new 16 32 8
color 1 #c68b67 skin
color 2 #3a2512 hair
color 3 #00a8a8 shirt
color 4 #3b3f9e pants
color 5 #555555 shoes
color 6 #f4f4f4 white
color 7 #4a3fc0 iris
color 8 #6b4528 mouth

# Body is the root. Each part is drawn while it is the active part, so its voxels join it.
part new Torso pivot 8 14 4
box 4 12 2 11 23 5 shirt

part new Head parent Torso pivot 8 24 4
box 4 24 0 11 31 7 skin
box 4 30 0 11 31 7 hair
box 4 24 0 11 29 0 hair
box 4 28 0 4 29 3 hair
box 11 28 0 11 29 3 hair
set 5 27 7 white
set 6 27 7 iris
set 9 27 7 iris
set 10 27 7 white
box 6 25 7 9 25 7 mouth

part new UpperArm.L parent Torso pivot 14 22 4
box 12 18 2 15 23 5 skin
box 12 20 2 15 23 5 shirt
part new LowerArm.L parent UpperArm.L pivot 14 18 4
box 12 12 2 15 17 5 skin
part new UpperArm.R parent Torso pivot 2 22 4
box 0 18 2 3 23 5 skin
box 0 20 2 3 23 5 shirt
part new LowerArm.R parent UpperArm.R pivot 2 18 4
box 0 12 2 3 17 5 skin

part new UpperLeg.L parent Torso pivot 10 11 4
box 8 6 2 11 11 5 pants
part new LowerLeg.L parent UpperLeg.L pivot 10 6 4
box 8 0 2 11 5 5 pants
box 8 0 2 11 1 5 shoes
part new UpperLeg.R parent Torso pivot 6 11 4
box 4 6 2 7 11 5 pants
part new LowerLeg.R parent UpperLeg.R pivot 6 6 4
box 4 0 2 7 5 5 pants
box 4 0 2 7 1 5 shoes
part active none

# Joint limits (degrees). Elbows and knees bend one way, arms and legs swing, head turns.
part limit Head x -50 50
part limit Head y -80 80
part limit Head z -30 30
part limit UpperArm.L x -180 90
part limit UpperArm.L z -10 180
part limit UpperArm.R x -180 90
part limit UpperArm.R z -180 10
part limit LowerArm.L x -150 0
part limit LowerArm.R x -150 0
part limit UpperLeg.L x -120 45
part limit UpperLeg.L z -10 45
part limit UpperLeg.R x -120 45
part limit UpperLeg.R z -45 10
part limit LowerLeg.L x 0 150
part limit LowerLeg.R x 0 150
rig check

# Poses. Rest, then a wave: right arm up in front with the forearm raised, a loose stance.
pose reset
pose save rest
pose set Torso 0 -8 0
pose set Head -5 -22 0
pose set UpperArm.R -90 0 -85
pose set LowerArm.R -90 0 0
pose set UpperArm.L 15 0 0
pose set LowerArm.L -10 0 0
pose set UpperLeg.L -12 0 0
pose set UpperLeg.R 8 0 0
pose set LowerLeg.R 15 0 0
pose save wave_a
pose set UpperArm.R -90 0 -85
pose set LowerArm.R -60 0 0
pose set Head -5 -28 4
pose save wave_b
clip new wave 0.8 loop wave_a wave_b
pose load wave_a
