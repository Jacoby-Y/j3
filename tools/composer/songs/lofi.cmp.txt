# Lofi loop: 8 bars at 64 bpm = 30 s, swung 8ths. ii-V-I-vi, then ii-V-iii-V/ii back to the top.
# Rebuild: rm songs/lofi.j3song; j3c-composer run songs/lofi --script songs/lofi.cmp.txt analyze
# Game loop: j3c-composer run songs/lofi --wav songs/lofi-loop.wav --loop
song tempo=64 key=C meter=4 bars=8 swing=0.64 noise=0.3
chords Dm9 G9 Cmaj7 Am7 Dm9 G9 Em7 A7
section A bars=1-4
section B bars=5-8

# Rootless voice-led chords: a hit on 1 held, another on the and of 3.
track Keys synth=keys vol=0.8 pan=-0.2 tone=0.35 reverb=0.3
fill Keys voiced rhythm=x__..x_. oct=3 v=72

track Bass synth=bass vol=0.85 tone=0.3
note Bass 1:1 2.5 D2 90
note Bass 1:4 1 A2 75
note Bass 2:1 2.5 G2 90
note Bass 2:4 1 D2 75
note Bass 3:1 2.5 C2 90
note Bass 3:4 1 B1 75
note Bass 4:1 2.5 A1 90
note Bass 4:4 1 C2 75
copy Bass bars=1-2 to=5
note Bass 7:1 2.5 E2 90
note Bass 7:4 1 B1 75
note Bass 8:1 2.5 A1 90
note Bass 8:4 1 C#2 80

# 16 steps a bar; swing pushes the off-beats late. Hats: 8ths with ghost 16ths (digits are velocity).
track Drums synth=drums vol=0.75 tone=0.25 drive=0.2
beat Drums kick x.........x..x.. bars=1-8 v=95
beat Drums snare ....x.......x... bars=1-8 v=80
beat Drums hat 4.3.4.324.3.4.32 bars=1-8
beat Drums kick x.......x.x..... bars=4
beat Drums kick x.........x..... bars=8
beat Drums snare ....x.......xxx. bars=8 v=70
beat Drums ohat ..............3. bars=8

# The melody comes in for the second half, so the loop has a lift.
track Melody synth=pluck vol=1.3 pan=0.25 tone=0.4 reverb=0.35
notes Melody 5:1 1/2 - - A4 _ C5 _ E5 _ v=80
notes Melody 6:1 1/2 D5 _ _ _ B4 _ A4 _ v=80
notes Melody 7:1 1/2 G4 _ _ _ - - B4 D5 v=80
notes Melody 8:1 1/2 E5 _ C#5 _ _ _ - - v=80

humanize Keys v=6 time=0.015
humanize Bass v=5 time=0.015
humanize Melody v=6 time=0.015
humanize Drums v=6
