# Stage order (README table, plan 11 A9/B1/B2). One stage per line, top to bottom:
#   NAME GLOB [GLOB...]
# GLOBs are relative to src/ and are shell patterns (`*` also crosses `/`). A file in a stage may
# `use j3c.X` only for X in its own stage or a stage above it. Every .j3 under src/ matches one line.
# Blank lines and `#` comments are skipped. Readers turn off pathname expansion (`set -f`) before
# splitting a line. tools/layers.sh checks it.
# project and ui sit above load (the loader builds .j3ui files and reads deps); ir sits above lower
# (lower builds IR; sema does not see IR, it owns `Ty` and lower maps it).
base base/*
project project/*
syntax syntax/*
ui ui.j3 ui/*
load load.j3 load/*
sema sema.j3 sema/*
ir ir.j3 ir/*
lower lower.j3 lower/*
devmode devmode.j3
shader shader.j3 shader/*
backend backend.j3 backend/*
# One stage: incremental objects run clang, and the clang driver keeps its stamps in the cache.
toolchain toolchain.j3 toolchain/* cache.j3 cache/*
driver driver.j3
tools cmd/* pkgtool.j3 doc.j3 doc/* lsp.j3 lsp/* mcp.j3 mcp/* scene.j3 scene/* editor.j3 editor/* assets/* dist.j3 dist/* textmate.j3 textmate/*
main main.j3
