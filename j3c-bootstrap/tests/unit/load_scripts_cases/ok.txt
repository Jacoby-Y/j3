// Made by j3c from the project's scenes (never written to disk).
module j3.scene3d.scripts;

use lib "c" as libc {
    fn getenv(name: cstr): cstr;
    fn strlen(s: cstr): int;
}

use j3.scene3d;
use j3.scene3d.{ Scene, NodeId };
use j3.math.Vec3;
use j3.memory.Arena;
use j3.option.Option;
use j3.text.{ cstr, same };

/// The scene set in j3.editor.toml.
pub const MAIN_SCENE: string = "scenes/main.j3scene";

/// The scene to run: the one the editor's Play picked (`J3_SCENE`), else `MAIN_SCENE`.
pub fn scene(): string {
    let raw = libc.getenv(cstr("J3_SCENE"));
    if raw == 0 { return MAIN_SCENE; }
    let len = libc.strlen(raw);
    if len < 1 { return MAIN_SCENE; }
    return []u8 { ptr: raw as &u8, len: len } as string;
}

const T__scripts__door = @script("scripts/door.j3s");
#[singleton]
let LIVE__scripts__door: []T__scripts__door;
#[singleton]
let IDS__scripts__door: []int;
#[singleton]
let COUNT__scripts__door: int;

const T__scripts__sub__lamp = @script("scripts/sub/lamp.j3s");
#[singleton]
let LIVE__scripts__sub__lamp: []T__scripts__sub__lamp;
#[singleton]
let IDS__scripts__sub__lamp: []int;
#[singleton]
let COUNT__scripts__sub__lamp: int;

fn own(set: int, id: int): int {
    if set == 0 { return id; }
    return set;
}

/// Make an instance for every object with a script, then run each `start`.
pub fn start(scene: &Scene) {
    COUNT__scripts__door = 0;
    COUNT__scripts__sub__lamp = 0;
    for k in 0..scene.len {
        let attached = scene.nodes[k].script;
        if same(attached, "scripts/door.j3s") { COUNT__scripts__door += 1; }
        if same(attached, "scripts/sub/lamp.j3s") { COUNT__scripts__sub__lamp += 1; }
    }
    LIVE__scripts__door = scene.arena.alloc::<T__scripts__door>(COUNT__scripts__door);
    IDS__scripts__door = scene.arena.alloc::<int>(COUNT__scripts__door);
    COUNT__scripts__door = 0;
    LIVE__scripts__sub__lamp = scene.arena.alloc::<T__scripts__sub__lamp>(COUNT__scripts__sub__lamp);
    IDS__scripts__sub__lamp = scene.arena.alloc::<int>(COUNT__scripts__sub__lamp);
    COUNT__scripts__sub__lamp = 0;
    for i in 0..scene.len {
        let id = scene.nodes[i].id;
        let path = scene.nodes[i].script;
        if same(path, "scripts/door.j3s") { spawn__scripts__door(scene, id); }
        if same(path, "scripts/sub/lamp.j3s") { spawn__scripts__sub__lamp(scene, id); }
    }
    for i in 0..COUNT__scripts__door { LIVE__scripts__door[i].start(scene); }
}

/// Run every instance's `update`.
pub fn update(scene: &Scene, dt: f32) {
    for i in 0..COUNT__scripts__door { LIVE__scripts__door[i].update(scene, dt); }
    for i in 0..COUNT__scripts__sub__lamp { LIVE__scripts__sub__lamp[i].update(scene, dt); }
}

fn spawn__scripts__door(scene: &Scene, id: int) {
    LIVE__scripts__door[COUNT__scripts__door] = T__scripts__door { node: NodeId { id: own(scene3d.object(scene, id, "node"), id) }, target: NodeId { id: scene3d.object(scene, id, "target") }, speed: scene3d.number(scene, id, "speed", 1.0), neg: scene3d.number(scene, id, "neg", -0.5), big: scene3d.number(scene, id, "big", 1e20) as f64, tiny: scene3d.number(scene, id, "tiny", 1e-5) as f64, whole: scene3d.number(scene, id, "whole", 3.0) as f64, count: scene3d.integer(scene, id, "count", -3) as int, small: scene3d.integer(scene, id, "small", 97) as u8, on: scene3d.flag(scene, id, "on", true), label: scene3d.label(scene, id, "label", "a \"b\"\tc"), dir: scene3d.vector(scene, id, "dir", Vec3.xyz(0, 0, 0)), plain: scene3d.number(scene, id, "plain", 0.0) };
    IDS__scripts__door[COUNT__scripts__door] = id;
    COUNT__scripts__door += 1;
}

/// `scene.script<...>(node)` for scripts/door.j3s.
pub fn find__scripts__door(scene: &Scene, node: NodeId): Option<&T__scripts__door> {
    for i in 0..COUNT__scripts__door {
        if IDS__scripts__door[i] == node.id { return Option.Some(&LIVE__scripts__door[i]); }
    }
    return Option.None;
}

fn spawn__scripts__sub__lamp(scene: &Scene, id: int) {
    LIVE__scripts__sub__lamp[COUNT__scripts__sub__lamp] = T__scripts__sub__lamp { glow: scene3d.number(scene, id, "glow", 2.5) };
    IDS__scripts__sub__lamp[COUNT__scripts__sub__lamp] = id;
    COUNT__scripts__sub__lamp += 1;
}

/// `scene.script<...>(node)` for scripts/sub/lamp.j3s.
pub fn find__scripts__sub__lamp(scene: &Scene, node: NodeId): Option<&T__scripts__sub__lamp> {
    for i in 0..COUNT__scripts__sub__lamp {
        if IDS__scripts__sub__lamp[i] == node.id { return Option.Some(&LIVE__scripts__sub__lamp[i]); }
    }
    return Option.None;
}
